package main

import (
	"context"
	"errors"
	"go-users-auth-template/internal/users"
	"log"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"github.com/go-chi/chi/v5"
	"github.com/go-chi/chi/v5/middleware"
)

func main() {
	r := chi.NewRouter()
	r.Use(middleware.Logger)

	// Health check endpoint
	r.Get("/health", func(w http.ResponseWriter, r *http.Request) {
		w.Write([]byte("Api working correctly"))
	})

	userHandlers := users.NewUserHandlers(users.NewUserService(users.NewUserInMemoryRepository()))

	r.Mount("/users", userHandlers.Routes())

	server := &http.Server{
		Addr:    ":8021",
		Handler: r,
	}

	// channel for waiting exit
	signalChan := make(chan os.Signal, 1)
	signal.Notify(signalChan, syscall.SIGINT, syscall.SIGTERM)

	go func() {
		slog.Info("server starting on port 8021 [http://127.0.0.1:8021/health]", "address", server.Addr)

		if err := http.ListenAndServe(":8021", r); err != nil && !errors.Is(err, http.ErrServerClosed) {
			log.Fatalf("Error starting server: %s", err)
			os.Exit(1)
		}
	}()

	/*
		shutdown section
	*/
	<-signalChan

	slog.Info("server shutting down")
	shutdownCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	if err := server.Shutdown(shutdownCtx); err != nil {
		slog.Error("server shutdown failer", "error", err)
		os.Exit(1)
	}
	slog.Info("server stopped")
}

package httpx

import (
	"encoding/json"
	"net/http"
)

func WriteJson(w http.ResponseWriter, status int, body any) error {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	return json.NewEncoder(w).Encode(body)
}

func WriteError(w http.ResponseWriter, status int, message string) {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)

	_ = json.NewEncoder(w).Encode(map[string]string{
		"error": message,
	})
}

package users

import (
	"encoding/json"
	"errors"
	"go-users-auth-template/internal/httpx"
	"net/http"
	"strconv"

	"github.com/go-chi/chi/v5"
)

// Define router which contain all user-logic endpoint
func (uh *UserHandlers) Routes() chi.Router {
	r := chi.NewRouter()

	r.Get("/", uh.getAllUsers)
	r.Post("/", uh.registerUser)

	r.Route("/{id}", func(r chi.Router) {
		r.Get("/", uh.getUserProfile)
		r.Put("/", uh.updateUserProfile)
		r.Delete("/", uh.deleteUser)
	})

	return r
}

type UserServiceAPI interface {
	RegisterUser(nickname, email, password string) (*User, error)
	GetAllUserProfiles() ([]*User, error)
	GetUserProfile(id int) (*User, error)
	UpdateUserProfile(user User) error
	DeleteUserProfile(id int) error
}

type UserHandlers struct {
	us *UserService
}

func NewUserHandlers(us *UserService) *UserHandlers {
	return &UserHandlers{us}
}

func (uh *UserHandlers) registerUser(w http.ResponseWriter, r *http.Request) {
	var reg RegisterUserRequest
	if err := json.NewDecoder(r.Body).Decode(&reg); err != nil {
		handleError(w, err)
		return
	}

	user, err := uh.us.RegisterUser(reg.Nickname, reg.Email, reg.Password)

	if err != nil {
		handleError(w, err)
		return
	}

	if err := httpx.WriteJson(w, http.StatusOK, user); err != nil {
		httpx.LogError("failer to create user while register", err, r)
	}
}

func (uh *UserHandlers) getAllUsers(w http.ResponseWriter, r *http.Request) {
	if users, err := uh.us.GetAllUserProfiles(); err != nil {
		handleError(w, err)
	} else {
		if err := httpx.WriteJson(w, http.StatusOK, users); err != nil {
			httpx.LogError("failed to write HTTP response", err, r)
		}
	}
}

func (uh *UserHandlers) getUserProfile(w http.ResponseWriter, r *http.Request) {
	id, err := handleUserID(r)
	if err != nil {
		handleError(w, ErrUserIDIdentifier)
		return
	}

	user, err := uh.us.GetUserProfile(id)
	if err != nil {
		handleError(w, err)
		return
	}

	if err := httpx.WriteJson(w, http.StatusOK, user); err != nil {
		httpx.LogError("failed to write HTTP response", err, r)
	}
}

func (uh *UserHandlers) updateUserProfile(w http.ResponseWriter, r *http.Request) {
	id, err := handleUserID(r)
	if err != nil {
		handleError(w, ErrUserIDIdentifier)
		return
	}

	var userUpdatingData UserUpdateRequest
	if err := json.NewDecoder(r.Body).Decode(&userUpdatingData); err != nil {
		handleError(w, ErrInvalidRequestFormat)
		return
	}

	if err := uh.us.UpdateUserProfile(id, userUpdatingData); err != nil {
		handleError(w, err)
		return
	}

	if err := httpx.WriteJson(w, http.StatusOK, userUpdatingData); err != nil {
		httpx.LogError("failed to write HTTP response", err, r)
	}
}

func (uh *UserHandlers) deleteUser(w http.ResponseWriter, r *http.Request) {
	id, err := handleUserID(r)
	if err != nil {
		handleError(w, ErrUserIDIdentifier)
		return
	}

	if err := uh.us.DeleteUserProfile(id); err != nil {
		handleError(w, err)
		return
	}

	if err := httpx.WriteJson(w, http.StatusOK, "Profile deleted successfully"); err != nil {
		httpx.LogError("failed to write HTTP response", err, r)
	}
}

func handleError(w http.ResponseWriter, err error) {
	switch {
	case errors.Is(err, ErrNotFound):
		httpx.WriteError(w, http.StatusNotFound, "user note found")
	case errors.Is(err, ErrInvalidRequestFormat):
		httpx.WriteError(w, http.StatusBadRequest, "invalid request")
	case errors.Is(err, ErrInvalidEmail):
		httpx.WriteError(w, http.StatusBadRequest, "Invalid user email")
	case errors.Is(err, ErrEmailTaken):
		httpx.WriteError(w, http.StatusBadRequest, "Email already taken")
	default:
		httpx.WriteError(w, http.StatusInternalServerError, "internal server error")
	}
}

func handleUserID(r *http.Request) (int, error) {
	strId := chi.URLParam(r, "id")
	return strconv.Atoi(strId)
}

package users

import "sync"

type UserInMemoryRepository struct {
	// here will be db connections pool, now map
	mu           sync.Mutex
	users        map[int]*User
	currentMaxId int
}

func NewUserInMemoryRepository() *UserInMemoryRepository {
	ur := &UserInMemoryRepository{
		users: make(map[int]*User),
	}
	ur.currentMaxId = 0
	return ur
}

func (ur *UserInMemoryRepository) CreateUser(nickname, email, password string) (*User, error) {
	userId := ur.currentMaxId
	ur.mu.Lock()
	ur.currentMaxId++
	ur.mu.Unlock()
	ur.users[userId] = &User{
		ID:       userId,
		Nickname: nickname,
		Email:    email,
		Password: password,
	}

	return ur.users[userId], nil
}

func (ur *UserInMemoryRepository) GetUserById(id int) (*User, error) {
	if user, ok := ur.users[id]; ok {
		return user, nil
	}

	return nil, ErrNotFound
}

func (ur *UserInMemoryRepository) GetAllUsers() ([]*User, error) {
	var users = make([]*User, 0, len(ur.users))
	for _, user := range ur.users {
		users = append(users, user)
	}
	return users, nil
}

func (ur *UserInMemoryRepository) UpdateUser(id int, nickname, email string) error {
	oldUser, ok := ur.users[id]
	if !ok {
		return ErrNotFound
	}
	ur.users[id] = &User{
		ID:       id,
		Nickname: nickname,
		Email:    email,
		Password: oldUser.Password,
	}
	return nil
}

func (ur *UserInMemoryRepository) DeleteUser(id int) error {
	if _, ok := ur.users[id]; !ok {
		return ErrNotFound
	}
	delete(ur.users, id)
	return nil
}

package users

import "fmt"

type UserRepository interface {
	CreateUser(nickname, email, password string) (*User, error)
	GetAllUsers() ([]*User, error)
	GetUserById(id int) (*User, error)
	UpdateUser(id int, nickname, email string) error
	DeleteUser(id int) error
}

type UserService struct {
	ur UserRepository
}

func NewUserService(ur UserRepository) *UserService {
	return &UserService{ur}
}

// method will be called only by auth microsercie after email confirmation
func (us *UserService) RegisterUser(nickname, email, password string) (*User, error) {
	if user, err := us.ur.CreateUser(nickname, email, password); err != nil {
		return nil, fmt.Errorf("error creating user: %v", err)
	} else {
		return user, nil
	}
}

func (us *UserService) GetUserProfile(id int) (*User, error) {
	return us.ur.GetUserById(id)
}

func (us *UserService) GetAllUserProfiles() ([]*User, error) {
	if users, err := us.ur.GetAllUsers(); err != nil {
		return nil, fmt.Errorf("error creating user: %v", err)
	} else {
		return users, nil
	}
}

func (us *UserService) UpdateUserProfile(id int, user UserUpdateRequest) error {
	if err := us.ur.UpdateUser(id, user.Nickname, user.Email); err != nil {
		return fmt.Errorf("error updating user: %v", err)
	}

	return nil
}

func (us *UserService) DeleteUserProfile(id int) error {
	if err := us.ur.DeleteUser(id); err != nil {
		return fmt.Errorf("error deleting user: %v", err)
	}

	return nil
}
